cmake_minimum_required(VERSION 3.20)
project(cpp_triangular_arbitrage VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL 1.1.1 REQUIRED)
find_package(Threads REQUIRED)
find_path(BOOST_INCLUDE_DIR boost/beast.hpp
  HINTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/usr/include" REQUIRED)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp
  HINTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/usr/include" REQUIRED)

add_library(market_core src/orderbook.cpp src/logger.cpp)
target_include_directories(market_core PUBLIC include)
target_include_directories(market_core SYSTEM PUBLIC ${JSON_INCLUDE_DIR})
target_compile_options(market_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(market_core PUBLIC Threads::Threads)
add_library(market_receiver src/receiver.cpp)
target_include_directories(market_receiver SYSTEM PUBLIC ${BOOST_INCLUDE_DIR})
target_compile_definitions(market_receiver PUBLIC BOOST_ERROR_CODE_HEADER_ONLY)
target_compile_options(market_receiver PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(market_receiver PUBLIC market_core OpenSSL::SSL OpenSSL::Crypto)
add_library(market_execution src/execution/execution.cpp src/execution/paper_gateway.cpp src/execution/binance_gateway.cpp)
target_include_directories(market_execution SYSTEM PUBLIC ${BOOST_INCLUDE_DIR})
target_compile_definitions(market_execution PUBLIC BOOST_ERROR_CODE_HEADER_ONLY)
target_compile_options(market_execution PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(market_execution PUBLIC market_core OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
add_executable(json_receiver src/main.cpp)
target_link_libraries(json_receiver PRIVATE market_receiver market_execution)

include(CTest)
if(BUILD_TESTING)
  add_executable(execution_tests tests/execution_tests.cpp)
  target_link_libraries(execution_tests PRIVATE market_execution)
  add_test(NAME execution COMMAND execution_tests)
  set_tests_properties(execution PROPERTIES TIMEOUT 20)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE market_core)
  add_test(NAME orderbook COMMAND core_tests)
  add_executable(logging_tests tests/logging_tests.cpp)
  target_link_libraries(logging_tests PRIVATE market_core)
  add_test(NAME logging COMMAND logging_tests)
  set_tests_properties(logging PROPERTIES TIMEOUT 20)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME websocket_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/websocket_integration.py $<TARGET_FILE:json_receiver>)
  set_tests_properties(websocket_integration PROPERTIES TIMEOUT 30)
endif()
